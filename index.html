<script>
const GRID = 3;

const board = document.getElementById("board");
const tray = document.getElementById("tray");
const progress = document.getElementById("progress");
const success = document.getElementById("success");

const IMAGE = "Screenshot-2024-04-10-at-12.25.19 PM-dc32c0352456381d-1024x454.png";

let BOARD_SIZE = board.clientWidth;
let PIECE_SIZE = BOARD_SIZE / GRID;

let activePiece = null;
let offsetX = 0;
let offsetY = 0;
let placed = 0;

function createGrid(){

for(let r = 0; r < GRID; r++){
for(let c = 0; c < GRID; c++){

const cell = document.createElement("div");

cell.className = "cell";

cell.style.width = PIECE_SIZE + "px";
cell.style.height = PIECE_SIZE + "px";

cell.style.left = (c * PIECE_SIZE) + "px";
cell.style.top = (r * PIECE_SIZE) + "px";

board.appendChild(cell);
}
}
}

function createPieces(){

const positions = [];

for(let r = 0; r < GRID; r++){
for(let c = 0; c < GRID; c++){
positions.push({r, c});
}
}

positions.sort(() => Math.random() - 0.5);

positions.forEach(pos => {

const piece = document.createElement("div");

piece.className = "piece";

piece.dataset.row = pos.r;
piece.dataset.col = pos.c;

piece.style.backgroundImage = `url('${IMAGE}')`;
piece.style.backgroundSize = `${BOARD_SIZE}px ${BOARD_SIZE}px`;

piece.style.backgroundPosition =
`-${pos.c * PIECE_SIZE}px -${pos.r * PIECE_SIZE}px`;

const angle = Math.floor(Math.random() * 40) - 20;
piece.style.transform = `rotate(${angle}deg)`;

tray.appendChild(piece);

piece.addEventListener("mousedown", startDrag);
piece.addEventListener("touchstart", startTouch, { passive:false });
});
}

function startDrag(e){

if(this.classList.contains("placed")) return;

activePiece = this;

activePiece.classList.add("dragging");

const rect = activePiece.getBoundingClientRect();

offsetX = e.clientX - rect.left;
offsetY = e.clientY - rect.top;

document.body.appendChild(activePiece);

activePiece.style.position = "fixed";

movePiece(e.clientX, e.clientY);

document.addEventListener("mousemove", moveMouse);
document.addEventListener("mouseup", endMouse);
}

function startTouch(e){

e.preventDefault();

const touch = e.touches[0];

startDrag.call(this, touch);

document.addEventListener("touchmove", moveTouch, { passive:false });
document.addEventListener("touchend", endTouch);
}

function moveMouse(e){
movePiece(e.clientX, e.clientY);
}

function moveTouch(e){

e.preventDefault();

const touch = e.touches[0];

movePiece(touch.clientX, touch.clientY);
}

function movePiece(x, y){

if(!activePiece) return;

activePiece.style.left = (x - offsetX) + "px";
activePiece.style.top = (y - offsetY) + "px";
}

function endMouse(e){

dropPiece(e.clientX, e.clientY);

document.removeEventListener("mousemove", moveMouse);
document.removeEventListener("mouseup", endMouse);
}

function endTouch(e){

const touch = e.changedTouches[0];

dropPiece(touch.clientX, touch.clientY);

document.removeEventListener("touchmove", moveTouch);
document.removeEventListener("touchend", endTouch);
}

function dropPiece(x, y){

if(!activePiece) return;

const rect = board.getBoundingClientRect();

const bx = x - rect.left;
const by = y - rect.top;

if(bx >= 0 && by >= 0 && bx <= BOARD_SIZE && by <= BOARD_SIZE){

const col = Math.floor(bx / PIECE_SIZE);
const row = Math.floor(by / PIECE_SIZE);

const correctRow = parseInt(activePiece.dataset.row);
const correctCol = parseInt(activePiece.dataset.col);

const alreadyPlaced = document.querySelector(
`.piece.placed[data-row='${row}'][data-col='${col}']`
);

if(row === correctRow && col === correctCol && !alreadyPlaced){

board.appendChild(activePiece);

activePiece.style.position = "absolute";
activePiece.style.left = (col * PIECE_SIZE) + "px";
activePiece.style.top = (row * PIECE_SIZE) + "px";
activePiece.style.transform = "rotate(0deg)";

activePiece.classList.add("placed");

placed++;

progress.innerText = `${placed} / 9 pieces placed`;

if(placed === GRID * GRID){
success.style.display = "flex";
}

} else {
returnToTray(activePiece);
}

} else {
returnToTray(activePiece);
}

activePiece.classList.remove("dragging");
activePiece = null;
}

function returnToTray(piece){

tray.appendChild(piece);

piece.style.position = "relative";
piece.style.left = "0px";
piece.style.top = "0px";

const angle = Math.floor(Math.random() * 40) - 20;

piece.style.transform = `rotate(${angle}deg)`;
}

createGrid();
createPieces();

</script>
